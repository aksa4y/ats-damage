cmake_minimum_required(VERSION 3.24)
project(ATSDamage VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ATS_DAMAGE_BUILD_PLUGIN "Build SPF plugin (Windows x64 for use in ATS)" ON)
set(SPF_ROOT "" CACHE PATH "SPF source checkout containing include/SPF/SPF_API")
add_library(damage_core STATIC src/DamageSystem.cpp)
target_include_directories(damage_core PUBLIC include)
set_target_properties(damage_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ATS_DAMAGE_BUILD_PLUGIN)
    if(NOT EXISTS "${SPF_ROOT}/include/SPF/SPF_API/SPF_Plugin.h")
        message(FATAL_ERROR "Set SPF_ROOT to the pinned SPF checkout; see README.md")
    endif()
    if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "ATS requires a 64-bit build")
    endif()
    add_library(ATSDamage SHARED src/main.cpp src/UI.cpp)
    target_link_libraries(ATSDamage PRIVATE damage_core)
    target_include_directories(ATSDamage PRIVATE "${SPF_ROOT}/include")
    set_target_properties(ATSDamage PROPERTIES PREFIX "")
    if(MSVC)
        target_compile_options(ATSDamage PRIVATE /utf-8 /W4)
    else()
        target_compile_options(ATSDamage PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS ATSDamage RUNTIME DESTINATION spfPlugins/ATSDamage)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(damage_tests tests/DamageSystemTests.cpp)
    target_link_libraries(damage_tests PRIVATE damage_core)
    add_test(NAME damage_logic COMMAND damage_tests)
endif()
